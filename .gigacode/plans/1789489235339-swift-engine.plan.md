## 5. Очередь запросов (RequestQueue)

### Интерфейс

```csharp
public interface IQueuedRequest
{
    string Id { get; }
    UniTask Execute();
}

public interface IQueuedRequest<T> : IQueuedRequest
{
    UniTask<T> Execute();
}
```

### Реализация

```csharp
public class RequestQueue
{
    private readonly Queue<(IQueuedRequest Request, Action OnComplete, Action<Exception> OnError)> _queue = new();
    private bool _isProcessing;

    public void Enqueue(IQueuedRequest request, Action onComplete = null, Action<Exception> onError = null)
    {
        _queue.Enqueue((request, onComplete, onError));
        ProcessNext();
    }

    private async void ProcessNext()
    {
        if (_isProcessing || _queue.Count == 0) return;
        _isProcessing = true;

        while (_queue.Count > 0)
        {
            var (request, onComplete, onError) = _queue.Dequeue();
            try
            {
                await request.Execute();
                onComplete?.Invoke();
            }
            catch (Exception ex)
            {
                onError?.Invoke(ex);
            }
        }
        _isProcessing = false;
    }
}
```

### 6b. EnergyModel

```csharp
public class EnergyModel
{
    private readonly GameConfig _config;
    private readonly ClickerModel _clickerModel;
    private int _currentEnergy;

    public int CurrentEnergy => _currentEnergy;
    public int MaxEnergy => _config.MaxEnergy;
    public event Action<int> OnEnergyChanged;

    public EnergyModel(GameConfig config, ClickerModel clickerModel)
    {
        _config = config;
        _clickerModel = clickerModel;
        _currentEnergy = _config.MaxEnergy;
        OnEnergyChanged?.Invoke(_currentEnergy);
        StartEnergyRefillLoop();
    }

    private async void StartEnergyRefillLoop()
    {
        while (true)
        {
            await UniTask.Delay(TimeSpan.FromSeconds(_config.EnergyRefillInterval));
            RefillEnergy();
        }
    }

    private void RefillEnergy()
    {
        _currentEnergy = Mathf.Min(_currentEnergy + _config.EnergyRefillAmount, _config.MaxEnergy);
        OnEnergyChanged?.Invoke(_currentEnergy);
    }

    public bool TrySpendEnergy(int amount)
    {
        if (_currentEnergy >= amount)
        {
            _currentEnergy -= amount;
            OnEnergyChanged?.Invoke(_currentEnergy);
            return true;
        }
        return false;
    }
}
```

### 6c. ClickerPresenter

```csharp
public class ClickerPresenter : IInitializable
{
    private readonly ClickerModel _clickerModel;
    private readonly EnergyModel _energyModel;
    private readonly ClickerView _view;
    private readonly GameConfig _config;

    public void Initialize()
    {
        _view.OnButtonClicked += OnButtonClick;
        _energyModel.OnEnergyChanged += OnEnergyChanged;

        // Auto-collect loop
        StartAutoCollectLoop();
    }

    private async void StartAutoCollectLoop()
    {
        if (!_config.AutoCollectEnabled) return;
        while (true)
        {
            await UniTask.Delay(TimeSpan.FromSeconds(_config.AutoCollectInterval));
            PerformClick(true); // true = auto
        }
    }

    private void OnButtonClick(bool isAuto)
    {
        PerformClick(isAuto);
    }

    private void PerformClick(bool isAuto)
    {
        var energyCost = isAuto ? _config.AutoCollectEnergyCost : _config.TapEnergyCost;
        var reward = isAuto ? _config.AutoCollectReward : _config.TapReward;

        if (!_energyModel.TrySpendEnergy(energyCost)) return;

        _clickerModel.AddCurrency(reward);
        _view.TriggerVFX(reward);
    }

    private void OnEnergyChanged(int energy)
    {
        // Presenter может уведомить view через событие
    }
}
```

### 6d. ClickerView — VFX

```csharp
public class ClickerView : MonoBehaviour
{
    public event Action<bool> OnButtonClicked; // bool = isAuto

    [Inject] private ObjectPool<ParticleBurstEffect> _particlePool;
    [Inject] private ObjectPool<FloatingCurrencyText> _floatingTextPool;
    [SerializeField] private Button _button;
    [SerializeField] private RectTransform _buttonRect;
    [SerializeField] private AudioSource _soundSource;

    public void Setup()
    {
        _button.onClick.AddListener(() => OnButtonClicked?.Invoke(false));
    }

    public void TriggerVFX(int reward)
    {
        // 4.1 Particle burst
        var particle = _particlePool.GetObject();
        particle.transform.position = _buttonRect.position;
        particle.Play();

        // 4.2 Floating currency text
        var floatingText = _floatingTextPool.GetObject();
        floatingText.Init(reward.ToString(), _buttonRect.position);

        // 4.3 Button press animation (DOTween)
        DOTween.To(() => _buttonRect.localScale, x => _buttonRect.localScale = x,
            Vector3.one * 0.9f, 0.1f)
            .SetEase(Ease.OutQuad)
            .OnComplete(() =>
            {
                DOTween.To(() => _buttonRect.localScale, x => _buttonRect.localScale = x,
                    Vector3.one, 0.15f).SetEase(Ease.OutBack);
            });

        // 4.4 Sound
        _soundSource.PlayOneShot(_soundSource.clip);
    }
}
```

### 6e. FloatingCurrencyText

```csharp
public class FloatingCurrencyText : MonoBehaviour
{
    private TextMeshProUGUI _text;

    public void Init(string value, Vector3 startPos)
    {
        _text = GetComponent<TextMeshProUGUI>();
        _text.text = value;
        transform.position = startPos;

        DOTween.To(() => 0f, x => _text.alpha = x, 1f, 0f)
            .SetTarget(this)
            .OnUpdate(() => { /* DOTween doesn't support lambda getters easily, use custom tween */ })
            .SetEase(Ease.OutQuad);

        // Проще:
        var tween = transform.DOMoveY(startPos.y + 2f, 1f).SetEase(Ease.OutQuad);
        var fadeTween = DOTween.To(() => _text.canvasRenderer.alpha, x =>
        {
            var col = _text.color;
            col.a = x;
            _text.color = col;
        }, 0f, 1f);

        tween.SetEase(Ease.OutQuad);
        fadeTween.SetEase(Ease.InQuad);
    }
}
```

> **Примечание:** Для DOTween с TMP нужно использовать `DOTween.To` с getter/setter через `Color.a`.

## 10. Verification (Как тестировать)

1. **Tab Switching:** Запустить сцену, убедиться что 3 вкладки переключаются,只显示 один tab content at a time.
2. **Clicker:**
   - Нажать кнопку — валюта +1, энергия -1, VFX (частицы, летающий текст, анимация кнопки, звук)
   - Подождать 10 сек — энергия +10
   - Подождать 3 сек — автосбор (валюта +1, энергия -1, все VFX)
   - Энергия на нуле — клики не работают (проверка)
   - Энергия не превышает 1000 (проверка)
3. **ScriptableObject:** Изменить значения в GameConfig — убедиться что все значения применяются без пересборки кода.
4. **Request Queue:** Добавить несколько stub-запросов — убедиться что они выполняются последовательно.
5. **Adaptive UI:** Изменить разрешение рендера — UI должен корректно масштабироваться.
